"""
Database initialization and connection management for Bharathi Oil Mill
Supports both Supabase (PostgreSQL) and local PostgreSQL
"""

import psycopg2
import psycopg2.extras
import streamlit as st
from contextlib import contextmanager
import os


def get_connection():
    """Get database connection from Streamlit secrets or environment variables."""
    db_url = st.secrets.get("DATABASE_URL") or os.environ.get("DATABASE_URL")
    if db_url:
        conn = psycopg2.connect(db_url)
    else:
        conn = psycopg2.connect(
            host=st.secrets.get("DB_HOST", "localhost"),
            port=st.secrets.get("DB_PORT", 5432),
            database=st.secrets.get("DB_NAME", "bharathi_oil_mill"),
            user=st.secrets.get("DB_USER", "postgres"),
            password=st.secrets.get("DB_PASSWORD", "password"),
        )
    return conn


@contextmanager
def get_db():
    conn = get_connection()
    try:
        yield conn
        conn.commit()
    except Exception as e:
        conn.rollback()
        raise e
    finally:
        conn.close()


def init_db():
    """Initialize all database tables."""
    with get_db() as conn:
        cur = conn.cursor()

        # ── Services / Mill Services ──────────────────────────────────────────
        cur.execute("""
            CREATE TABLE IF NOT EXISTS services (
                id SERIAL PRIMARY KEY,
                category VARCHAR(50) NOT NULL,
                name VARCHAR(100) NOT NULL,
                rate_per_kg DECIMAL(10,2) NOT NULL,
                unit VARCHAR(20) DEFAULT 'kg',
                is_active BOOLEAN DEFAULT TRUE,
                created_at TIMESTAMP DEFAULT NOW(),

        # ── Customers ─────────────────────────────────────────────────────────

def generate_bill_number():
    """Generate sequential bill number like BOM-20241114-0001"""
    from datetime import date
    today = date.today().strftime("%Y%m%d")
    with get_db() as conn:
        cur = conn.cursor()
        cur.execute(
            "SELECT COUNT(*) FROM bills WHERE bill_number LIKE %s",
            (f"BOM-{today}-%",)
        )
        count = cur.fetchone()[0]
    return f"BOM-{today}-{count+1:04d}"

import os
from pathlib import Path

# Folder to scan
SOURCE_FOLDER = r"C:\Users\udhay\Downloads\bharathi_oil_mill\bharathi_oil_mill"

# Output file
OUTPUT_FILE = "project_code_dump.txt"

# Code file extensions
CODE_EXTENSIONS = {
    ".py", ".js", ".ts", ".tsx", ".jsx",
    ".java", ".cs", ".cpp", ".c", ".h",
    ".php", ".go", ".rs", ".swift",
    ".html", ".css", ".scss",
    ".sql", ".json", ".yaml", ".yml",
    ".xml", ".sh", ".bat", ".ps1",
    ".dart", ".kt", ".vue"
}

with open(OUTPUT_FILE, "w", encoding="utf-8") as outfile:

    outfile.write("=" * 120 + "\n")
    outfile.write("PROJECT CODE EXPORT\n")
    outfile.write("=" * 120 + "\n\n")

    for root, dirs, files in os.walk(SOURCE_FOLDER):

        # Ignore common folders
        dirs[:] = [
            d for d in dirs
            if d not in {
                ".git",
                ".venv",
                "venv",
                "__pycache__",
                "node_modules",
                ".next",
                "dist",
                "build"
            }
        ]

        for file in files:

            filepath = os.path.join(root, file)
            ext = Path(file).suffix.lower()

            if ext not in CODE_EXTENSIONS:
                continue

            try:
                relative_path = os.path.relpath(
                    filepath,
                    SOURCE_FOLDER
                )

                with open(
                    filepath,
                    "r",
                    encoding="utf-8",
                    errors="ignore"
                ) as f:
                    content = f.read()

                outfile.write("\n")
                outfile.write("=" * 120 + "\n")
                outfile.write(f"FILE: {file}\n")
                outfile.write(f"PATH: {filepath}\n")
                outfile.write(f"RELATIVE_PATH: {relative_path}\n")
                outfile.write(f"EXTENSION: {ext}\n")
                outfile.write("=" * 120 + "\n\n")

                outfile.write(content)
                outfile.write("\n\n")

            except Exception as e:
                outfile.write(
                    f"\nERROR READING {filepath}: {str(e)}\n"
                )

print(f"Export completed: {OUTPUT_FILE}")

    if not items:
        st.info("No inventory items found.")
    else:
        df = pd.DataFrame(items)
        df["value"] = df["quantity"] * df["rate_per_unit"]
        df["status"] = df.apply(
            lambda r: "🔴 Low" if r["quantity"] <= r["reorder_level"] and r["reorder_level"] > 0
                      else "✅ OK", axis=1)
        df["icon"] = df["category"].map(CAT_ICONS).fillna("📋")
        df["display"] = df["icon"] + " " + df["item_name"]

        # Summary cards
        total_val = df["value"].sum()
        low_stock = (df["quantity"] <= df["reorder_level"]).sum()
        c1, c2, c3 = st.columns(3)
        c1.metric("Total Inventory Value", f"₹{total_val:,.2f}")
        c2.metric("Total Items", len(df))
        c3.metric("Low Stock Alerts", int(low_stock), delta_color="inverse")

        st.markdown("---")
        
        for cat, group in df.groupby("category"):
            icon = CAT_ICONS.get(cat, "📋")
            st.markdown(f"#### {icon} {cat.replace('_', ' ').title()}")
            display_df = group[["item_name", "quantity", "unit", "rate_per_unit", "value", "status", "reorder_level"]].copy()
            display_df.columns = ["Item", "Qty", "Unit", "Rate/Unit (₹)", "Value (₹)", "Status", "Reorder Level"]
            display_df["Qty"] = display_df["Qty"].apply(lambda x: f"{x:.3f}")
            display_df["Rate/Unit (₹)"] = display_df["Rate/Unit (₹)"].apply(lambda x: f"₹{x:.2f}")
            display_df["Value (₹)"] = display_df["Value (₹)"].apply(lambda x: f"₹{x:.2f}")
            st.dataframe(display_df, use_container_width=True, hide_index=True)

STATUS_COLORS = {
    "active": "#27ae60",
    "maintenance": "#e67e22",
    "breakdown": "#e74c3c",
    "inactive": "#95a5a6",
}
STATUS_ICONS = {
    "active": "✅", "maintenance": "🔧", "breakdown": "🔴", "inactive": "⚫"
}
TYPE_ICONS = {
    "grain_milling": "🌾", "spice_grinding": "🌶", "dhal_splitting": "🫘",
    "oil_extraction": "🫒", "oil_cake_grinding": "🟤", "wet_grinding": "💧",
}

    ["🏭 Machine Fleet", "➕ Add Machine", "🔧 Log Maintenance", "📋 Maintenance History"])

with tab_fleet:
    with get_db() as conn:
        cur = conn.cursor(cursor_factory=psycopg2.extras.RealDictCursor)
        cur.execute("""
            SELECT m.*,
                   COALESCE(SUM(mm.cost), 0) as total_maint_cost,
                   MAX(mm.maintenance_date) as last_maint
            FROM machines m
            LEFT JOIN machine_maintenance mm ON mm.machine_id = m.id
            GROUP BY m.id ORDER BY m.machine_type, m.machine_name
        """)
        machines = cur.fetchall()

    # Summary
    total = len(machines)
    active = sum(1 for m in machines if m["status"] == "active")
    in_maint = sum(1 for m in machines if m["status"] == "maintenance")
    broken = sum(1 for m in machines if m["status"] == "breakdown")

    c1, c2, c3, c4 = st.columns(4)
    c1.metric("Total Machines", total)
    c2.metric("✅ Active", active)
    c3.metric("🔧 In Maintenance", in_maint)
    c4.metric("🔴 Breakdown", broken)

    st.markdown("---")

    # Group by type
    machine_by_type = {}
    for m in machines:
        machine_by_type.setdefault(m["machine_type"], []).append(m)

    for mtype, mlist in machine_by_type.items():
        icon = TYPE_ICONS.get(mtype, "⚙️")
        st.markdown(f"#### {icon} {mtype.replace('_', ' ').title()}")
        for m in mlist:
            status_col = STATUS_COLORS.get(m["status"], "#888")
            status_icon = STATUS_ICONS.get(m["status"], "⚙️")
            days_to_service = None
            overdue = False
            if m["next_service_date"]:
                days_to_service = (m["next_service_date"] - date.today()).days
                overdue = days_to_service < 0

            with st.container():
                col_name, col_status, col_maint, col_action = st.columns([3, 1.5, 2, 1.5])
                col_name.markdown(f"**{m['machine_name']}**  \n"
                                  f"<small>S/N: {m.get('serial_number') or '—'}</small>",
                                  unsafe_allow_html=True)
                col_status.markdown(
                    f'<span style="color:{status_col};font-weight:600">{status_icon} {m["status"].upper()}</span>',
                    unsafe_allow_html=True)
                if days_to_service is not None:
                    svc_text = f"⚠️ **OVERDUE {abs(days_to_service)}d**" if overdue else f"Service in {days_to_service}d"
                    col_maint.markdown(svc_text)
                else:
                    col_maint.write("—")
                new_status = col_action.selectbox("", list(STATUS_COLORS.keys()),
                                                  index=list(STATUS_COLORS.keys()).index(m["status"]),
                                                  key=f"sts_{m['id']}")
                if new_status != m["status"]:
                    with get_db() as conn:
                        cur = conn.cursor()
                        cur.execute("UPDATE machines SET status=%s WHERE id=%s",
                                    (new_status, m["id"]))
                    st.rerun()

"""QR Code generation utilities for Bharathi Oil Mill bills."""

import qrcode
import json
import io
import base64
from datetime import datetime


def generate_bill_qr(bill_data: dict) -> str:
    """
    Generate a QR code for a bill and return as base64 PNG string.
    
    The QR encodes a compact JSON payload:
      bill_number, date, customer, total, items (name + qty + amount)
    """
    payload = {
        "app": "BharathaOilMill",
        "bill": bill_data.get("bill_number", ""),
        "date": bill_data.get("created_at", datetime.now()).strftime("%d-%m-%Y %H:%M") 
                if hasattr(bill_data.get("created_at", ""), "strftime") 
                else str(bill_data.get("created_at", "")),
        "customer": bill_data.get("customer_name") or "Walk-in",
        "phone": bill_data.get("customer_phone") or "",
        "total": float(bill_data.get("total", 0)),
        "payment": bill_data.get("payment_method") or "cash",
        "status": bill_data.get("payment_status") or "paid",
    }
    if bill_data.get("items"):
        payload["items"] = [
            {
                "s": item.get("service_name", ""),
                "q": float(item.get("quantity_kg", 0)),
                "a": float(item.get("amount", 0)),
            }
            for item in bill_data["items"]
        ]

    qr_text = json.dumps(payload, separators=(",", ":"))

    qr = qrcode.QRCode(
        version=None,
        error_correction=qrcode.constants.ERROR_CORRECT_M,
        box_size=6,
        border=2,
    )
    qr.add_data(qr_text)
    qr.make(fit=True)

    img = qr.make_image(fill_color="#1a1a2e", back_color="white")
    buf = io.BytesIO()
    img.save(buf, format="PNG")
    buf.seek(0)
    return base64.b64encode(buf.read()).decode("utf-8")


def qr_to_html_img(b64_string: str, size: int = 150) -> str:
    """Return an <img> tag with embedded base64 QR PNG."""
    return f'<img src="data:image/png;base64,{b64_string}" width="{size}" height="{size}" />'
